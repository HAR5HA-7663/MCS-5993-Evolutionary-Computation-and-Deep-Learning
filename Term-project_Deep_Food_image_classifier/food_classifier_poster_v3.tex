%% LEFT COLUMN
\begin{column}{0.32\textwidth}

\begin{tcolorbox}[posterbox, title=ABSTRACT]
\footnotesize
Compares three HPO methods: ES(1+1), Keras Tuner Hyperband, DEAP ($\mu$+$\lambda$)-ES for a 10-class American food classifier. Achieves \textbf{79.8\% accuracy}—\textbf{24\% improvement} over baseline. Uses dual RTX 3090 GPUs, 10,000 images, optimizes 5 hyperparameters.
\end{tcolorbox}

\vspace{1mm}

\begin{tcolorbox}[posterbox, title=INTRODUCTION]
\centering
\includegraphics[width=0.98\textwidth]{food_samples.png}\\
{\tiny Figure: 10 American food classes from Food-101}
\end{tcolorbox}

\vspace{1mm}

\begin{tcolorbox}[posterbox, title=DATASET]
\footnotesize
\textbf{Source:} Food-101 (10 classes) $\bullet$ 10,000 images $\bullet$ 224×224\\
\textbf{Split:} Train 70\% / Val 15\% / Test 15\%\\
\textbf{Augmentation:} Flip, Rotation, Zoom, Contrast
\end{tcolorbox}

\vspace{1mm}

\begin{tcolorbox}[posterbox, title=HPO METHODS]
\footnotesize
\textbf{Baseline:} Fixed LR=0.002, Dropout=0.25\\
\textbf{ES(1+1):} Gaussian mutation, 1/5 rule, 15 gen\\
\textbf{Keras Tuner:} Hyperband, 174 trials\\
\textbf{DEAP:} ($\mu$+$\lambda$)-ES, $\mu$=5, $\lambda$=10
\end{tcolorbox}

\end{column}

\begin{tcolorbox}[posterbox, title=KEY FINDINGS]
\footnotesize
$\bullet$ All HPO methods achieved $\sim$79\% accuracy\\
$\bullet$ ES(1+1) slightly outperformed library methods\\
$\bullet$ Label smoothing critical: 0.05 $\rightarrow$ 0.30\\
$\bullet$ Optimal LR: $\sim$1.4$\times$10$^{-3}$\\
$\bullet$ \textbf{24\% improvement} validates HPO importance
\end{tcolorbox}
