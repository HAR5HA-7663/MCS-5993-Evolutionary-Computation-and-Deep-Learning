\begin{block}{ABSTRACT}
This research compares three hyperparameter optimization (HPO) methods for deep learning-based food classification: Evolution Strategy (1+1), Keras Tuner Hyperband, and DEAP ($\mu$+$\lambda$)-ES. A custom ResNet-style CNN was trained on 10,000 images across 10 American food categories from the Food-101 dataset. All three HPO methods achieved approximately 79\% test accuracy, representing a 24\% improvement over the baseline model (64.37\%). The study utilized dual NVIDIA RTX 3090 GPUs (48GB VRAM) and optimized five critical hyperparameters: learning rate, dropout rate, label smoothing, rotation range, and zoom range. Results demonstrate that evolutionary strategies can be competitive with sophisticated library implementations for CNN hyperparameter tuning.
\end{block}

\begin{block}{INTRODUCTION}
\textbf{Background:} Deep learning has revolutionized computer vision, but optimal hyperparameter configurations remain challenging to discover manually. Evolutionary algorithms offer a biologically-inspired approach to navigate complex hyperparameter spaces.

\textbf{Problem Statement:} While sophisticated HPO libraries exist, understanding fundamental evolutionary strategies provides insight into optimization dynamics and enables customization for domain-specific constraints.

\textbf{Research Question:} How do simple evolutionary strategies like ES(1+1) compare against modern HPO frameworks (Keras Tuner Hyperband, DEAP) for food image classification tasks?

    \begin{itemize}[leftmargin=*,itemsep=0pt]

\begin{block}{METHODS}
\textbf{Dataset Preparation:}
\begin{itemize}[leftmargin=*,itemsep=0pt,topsep=2pt]
    \item Food-101 subset: 10 American food classes
    \item 10,000 total images (1,000 per class)
    \item Split: 70\% training / 15\% validation / 15\% test
    \item Preprocessing: $224 \times 224$ resize, normalization [0,1]
    \item Augmentation: horizontal flip, rotation, zoom, contrast

\textbf{Optimization Methods:}
\begin{itemize}[leftmargin=*,itemsep=0pt]
    \item \textbf{Baseline:} Fixed hyperparameters (LR=0.002, Dropout=0.25)
    \item \textbf{ES(1+1):} Single parent/offspring, Gaussian mutation, adaptive step size via 1/5 success rule, 15 generations
    \item \textbf{Keras Tuner:} Hyperband algorithm, factor=3, max\_epochs=3, 174 trials
    \item \textbf{DEAP ($\mu$+$\lambda$):} $\mu$=5 parents, $\lambda$=10 offspring, 10 generations

    \item Framework: TensorFlow 2.10 with MirroredStrategy
    \item Precision: Mixed precision (float16)
    \item Optimizer: Adam with cosine decay
    \item Epochs: 50 per model training
    \item Early stopping: patience=5, monitor validation loss
\end{itemize}

\begin{block}{MODEL \& SYSTEM DESIGN}
\textbf{CNN Architecture:} Custom ResNet-style network with residual connections

\begin{center}
\includegraphics[width=0.75\columnwidth]{cnn_architecture.png}
\captionof{figure}{ResNet-style CNN architecture with 4 residual blocks}
\end{center}

\textbf{Architecture Details:}
\begin{itemize}[leftmargin=*,itemsep=0pt]
    \item \textbf{Input Layer:} $224 \times 224 \times 3$ RGB images
    \item \textbf{Initial Convolution:} 64 filters, $7 \times 7$ kernel, stride=2
    \item \textbf{Residual Block 1:} 64 $\rightarrow$ 128 channels (2 conv layers)
    \item \textbf{Residual Block 2:} 128 $\rightarrow$ 256 channels (2 conv layers)
    \item \textbf{Residual Block 3:} 256 $\rightarrow$ 512 channels (2 conv layers)
    \item \textbf{Residual Block 4:} 512 $\rightarrow$ 1024 channels (2 conv layers)
    \item \textbf{Global Average Pooling:} Reduces spatial dimensions
    \item \textbf{Dense Layer:} 1024 units with ReLU activation
    \item \textbf{Dropout:} Regularization (rate optimized via HPO)
    \item \textbf{Output Layer:} 10 units with Softmax activation
    \item \textbf{Total Parameters:} $\sim$20.6 million
\end{itemize}

\textbf{Key Design Principles:}
\begin{itemize}[leftmargin=*,itemsep=0pt]
    \item \textbf{Residual Connections:} Skip connections enable gradient flow through deep networks, mitigating vanishing gradients
    \item \textbf{Batch Normalization:} Stabilizes training by normalizing layer inputs, enabling higher learning rates
    \item \textbf{Progressive Feature Extraction:} Channel depth increases (64$\rightarrow$1024) while spatial dimensions decrease via pooling
    \item \textbf{Global Average Pooling:} Reduces overfitting compared to fully-connected layers, fewer parameters
\end{itemize}

\textbf{Data Augmentation Pipeline:}
\begin{itemize}[leftmargin=*,itemsep=0pt]
    \item Random horizontal flip (50\% probability)
    \item Random rotation (range optimized via HPO)
    \item Random zoom (range optimized via HPO)
    \item Random contrast adjustment
    \item Purpose: Increases training diversity, improves generalization
\end{itemize}

\textbf{Evolutionary Strategy Implementation:}

\textbf{ES(1+1) Algorithm:}
\begin{enumerate}[leftmargin=*,itemsep=0pt]
    \item Initialize random hyperparameter vector $\mathbf{x}^{(0)}$
    \item Initialize step size $\sigma^{(0)} = 0.15$
    \item For generation $t = 1$ to $T$:
    \begin{enumerate}[label=\alph*)]
        \item Generate offspring: $\mathbf{x}' = \mathbf{x}^{(t-1)} + \mathcal{N}(\mathbf{0}, \sigma^{2}\mathbf{I})$
        \item Train CNN with $\mathbf{x}'$, evaluate fitness (validation accuracy)
        \item If $f(\mathbf{x}') \geq f(\mathbf{x}^{(t-1)})$: accept offspring, increment success counter
        \item Else: reject offspring, keep parent
        \item Every 5 generations: apply 1/5 success rule
        \begin{itemize}
            \item If success rate $> 1/5$: $\sigma \leftarrow 1.2\sigma$ (increase exploration)
            \item If success rate $< 1/5$: $\sigma \leftarrow 0.8\sigma$ (increase exploitation)
        \end{itemize}
    \end{enumerate}
    \item Return best hyperparameter configuration
\end{enumerate}

\textbf{Fitness Function:} $f(\mathbf{x}) = $ validation accuracy after 50 training epochs

\begin{block}{SUMMARY, DISCUSSION, and/or CONCLUSION}
\textbf{Summary of Findings:}
\begin{itemize}[leftmargin=*,itemsep=0pt]
    \item All three HPO methods achieved substantial improvements (23--24\%) over baseline CNN
    \item Simple ES(1+1) competitive with sophisticated library implementations
    \item Computational efficiency favors ES(1+1): $8\times$ fewer evaluations than DEAP
    \item Hyperparameter convergence consistency validates optimization robustness
\end{itemize}

\textbf{Critical Insights:}
\begin{enumerate}[leftmargin=*,itemsep=0pt]
    \item \textbf{Label Smoothing Impact:} Increasing from 0.05 to baseline 0.00 contributed significantly to improved generalization
    \item \textbf{Learning Rate Tuning:} Optimal LR ($1.4$--$1.9 \times 10^{-3}$) notably lower than baseline ($2.0 \times 10^{-3}$)
    \item \textbf{Regularization Balance:} Lower dropout + label smoothing more effective than aggressive dropout alone
    \item \textbf{Data Augmentation:} Moderate ranges prevent overfitting without excessive distortion
\end{enumerate}

\textbf{Practical Implications:}
\begin{itemize}[leftmargin=*,itemsep=0pt]
    \item ES(1+1) viable for resource-constrained HPO scenarios
    \item Custom evolutionary strategies enable domain-specific optimization constraints
    \item Multi-GPU training essential for acceptable HPO turnaround times
    \item Mixed precision reduces memory footprint without accuracy loss
\end{itemize}

\textbf{Limitations:}
\begin{itemize}[leftmargin=*,itemsep=0pt]
    \item Limited to 10 food classes (subset of Food-101's 101 classes)
    \item Fixed architecture: HPO focused on training hyperparameters only
    \item Computational budget constraints limited search iterations
    \item Single random seed per evaluation (no multiple-run averaging)
\end{itemize}

\textbf{Contributions:}
\begin{itemize}[leftmargin=*,itemsep=0pt]
    \item Empirical comparison of evolutionary vs. library HPO methods
    \item Demonstration of ES(1+1) competitiveness in deep learning context
    \item Open-source implementation on GitHub
    \item Deployed model on HuggingFace Hub for public use
\end{itemize}
\end{block}
